cmake_minimum_required(VERSION 3.20)
project(tab5_ui_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(APP "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_DIR "${APP}/managed_components/lvgl__lvgl" CACHE PATH "LVGL 9.2.2 source (resolved by IDF build)")
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(ui_smoke smoke.cpp "${APP}/main/ui.cpp" "${APP}/main/fonts/rf_jp20.c" "${APP}/main/fonts/rf_jp24.c")
target_include_directories(ui_smoke PRIVATE "${APP}/main" "${CMAKE_CURRENT_LIST_DIR}/stubs")
target_link_libraries(ui_smoke PRIVATE lvgl)
if(MSVC)
  target_compile_options(lvgl PRIVATE /utf-8)
  target_compile_options(ui_smoke PRIVATE /utf-8 /EHsc /UNDEBUG)
  target_compile_definitions(ui_smoke PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(ui_smoke PRIVATE -UNDEBUG)
  target_link_libraries(ui_smoke PRIVATE m)
endif()
enable_testing()
add_test(NAME portrait_ui COMMAND ui_smoke)
